Options -Indexes
AddDefaultCharset UTF-8
DirectoryIndex index.html

RewriteEngine On

# ── Forzar HTTPS ──────────────────────────────────────────────────
RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]

# ── GEO: servir index.md cuando un agente pide text/markdown ──────
RewriteCond %{HTTP_ACCEPT} text/markdown [NC]
RewriteCond %{REQUEST_URI} ^/?$
RewriteRule ^$ /index.md [L]

# ── Eliminar /index.html de URLs de directorio (301) ─────────────
RewriteCond %{THE_REQUEST} /index\.html[\s?] [NC]
RewriteRule ^(.*/)index\.html$ /$1 [R=301,L]

# ── Eliminar .html de cualquier otra página (301) ─────────────────
RewriteCond %{THE_REQUEST} \.html[\s?] [NC]
RewriteRule ^(.+)\.html$ /$1 [R=301,L]

# ── Servir el archivo .html internamente sin extensión ────────────
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME}.html -f
RewriteRule ^(.+)$ $1.html [L]

# ── Página de error 404 (status real, no soft-404) ─────────────────
ErrorDocument 404 /404.html

# ── Headers de seguridad ────────────────────────────────────────────
<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload"
  Header always set X-Frame-Options "DENY"
  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=(), usb=()"
  Header always set X-XSS-Protection "0"
  Header unset X-Powered-By

  # ── Performance: cache ──────────────────────────────────────────
  <FilesMatch "\.(css|js|jpg|jpeg|png|gif|ico|webp|avif|svg|woff2|woff)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>
  <FilesMatch "\.(html|htm|php)$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>

  # ── GEO / content negotiation ────────────────────────────────────
  <FilesMatch "\.(txt|xml|json)$">
    Header always set Content-Type "text/plain; charset=utf-8"
  </FilesMatch>
  <FilesMatch "\.md$">
    Header always set Content-Type "text/markdown; charset=utf-8"
  </FilesMatch>
  Header append Vary "Accept"
</IfModule>
